        end)
    end

        if not hrp then return end

        if (lookPos - currentPos).Magnitude > 0.08 then
            hrp.CFrame = CFrame.new(currentPos, lookPos)
        end
    end)
end

local function IsAptFree(board)
    local text = string.lower(GetOccupationText(board))
    if text == "" or text == "empty" or text == "for sale" or text == "available"
        or string.find(text, "vacant", 1, true) or string.find(text, "none", 1, true)
        or string.find(text, "unoccupied", 1, true) or string.find(text, "libre", 1, true)
        or string.find(text, "sale", 1, true) or string.find(text, "buy", 1, true)
        or text == "---" or text == "-" or text == "n/a" then
        return true
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if string.find(text, string.lower(plr.Name), 1, true)
            or string.find(text, string.lower(plr.DisplayName), 1, true) then
            return false
        end
    end
    return #text < 3
end

local function GetFrontOfBoardCFrame(boardPart)
    local look = boardPart.CFrame.LookVector

    local stand = boardPart.Position + front * 2.2
    stand = Vector3.new(stand.X, boardPart.Position.Y - 1.2, stand.Z)
    if stand.Y < 1 then stand = Vector3.new(stand.X, math.max(boardPart.Position.Y - 1.2, 3), stand.Z) end
    return CFrame.new(stand, boardPart.Position)
end

local function FindFreeCompactWalkup()
    local apartments = GetApartmentsFolder()
    if not apartments then return nil, "no_folder" end
    local candidates = {}
    for _, aptFolder in ipairs(apartments:GetChildren()) do
        if not IsAllowedAptFolder(aptFolder) then continue end
        local apt = aptFolder:FindFirstChild("Apartment") or aptFolder
        local board = GetBoard(apt) or GetBoard(aptFolder)
        if not board then continue end
        if not IsAptFree(board) then continue end
        local pots = GetCookingPots(apt)
        if #pots == 0 then pots = GetCookingPots(aptFolder) end
        if #pots == 0 then continue end
        local boardPart = board:IsA("BasePart") and board or board:FindFirstChildWhichIsA("BasePart", true)
        if not boardPart then continue end
        table.insert(candidates, {
            apt = apt, aptFolder = aptFolder, name = aptFolder.Name,
            board = board, boardPos = boardPart.Position, boardPart = boardPart,
            cf = GetFrontOfBoardCFrame(boardPart),
        })
    end
    if #candidates == 0 then return nil, "full" end
    return candidates[math.random(1, #candidates)], nil
end

local function IsProtectedObject(obj)
    if not obj then return true end
    if obj.Name == "SafePlatform" or obj:IsA("Terrain") then return true end
    if obj:IsA("ProximityPrompt") then return true end
    local n = string.lower(obj.Name or "")
    if n == "board" or string.find(n, "board", 1, true) then return true end
    if obj:FindFirstAncestor("Board") then return true end
    if n == "cooking pot" or n == "cookingpot" or string.find(n, "cooking", 1, true) then return true end
    if n == "lamont bell" or string.find(n, "lamont", 1, true) or string.find(n, "bell", 1, true) then return true end
    local ancestor = obj
    while ancestor and ancestor ~= workspace and ancestor ~= game do
        local an = string.lower(ancestor.Name or "")
        if an == "lamont bell" or string.find(an, "lamont", 1, true) then return true end
        if an == "npcs" or an == "npc" then return true end
        ancestor = ancestor.Parent
    end
    local model = obj:FindFirstAncestorWhichIsA("Model")
    if model and Players:GetPlayerFromCharacter(model) then return true end
    if obj:IsA("Model") and Players:GetPlayerFromCharacter(obj) then return true end
    if obj:IsA("Humanoid") or obj:FindFirstChildOfClass("Humanoid") then return true end
    return false
end

local function IsWall(obj)
    if not obj or not obj:IsA("BasePart") then return false end
    local n = string.lower(obj.Name or "")
    if n == "wall" or string.find(n, "wall", 1, true) or string.find(n, "pared", 1, true) then return true end
    if string.find(n, "fence", 1, true) or string.find(n, "barrier", 1, true) then return true end
    local s = obj.Size
    if s.Y > 4 and (s.X < 4 or s.Z < 4) then return true end
    if s.Y > 6 and s.X * s.Z < 50 then return true end
    return false
end

local function CloseAndLockDoor(apt, aptFolder)
    local root = apt or aptFolder
    if not root then
        SetStatus("Sin apt")
        return false
    end

    local doorsFolder = root:FindFirstChild("Doors") or root:FindFirstChild("Doors", true)
    local door = doorsFolder and doorsFolder:FindFirstChild("Door") or root:FindFirstChild("Door", true)
    if not door then
        SetStatus("No Door")
        return false
    end

    local doorAttach = door:FindFirstChild("DoorAttach")
    local lockAttach = door:FindFirstChild("LockAttach")
    local doorPrompt = doorAttach and doorAttach:FindFirstChildOfClass("ProximityPrompt")
    local lockPrompt = lockAttach and lockAttach:FindFirstChildOfClass("ProximityPrompt")
    if not doorPrompt and doorAttach then
        doorPrompt = doorAttach:FindFirstChildWhichIsA("ProximityPrompt", true)
    end
    if not lockPrompt and lockAttach then
        lockPrompt = lockAttach:FindFirstChildWhichIsA("ProximityPrompt", true)
    end

    local function lockText()
        return string.lower(tostring(lockPrompt and lockPrompt.ActionText or ""))
    end

    local function isLocked()
        return string.find(lockText(), "unlock", 1, true) ~= nil
    end

    local function needsLock()
        local t = lockText()
        if string.find(t, "unlock", 1, true) then return false end
        return t == "lock" or string.find(t, "lock", 1, true) ~= nil
    end

    local function getSwingPart()
        if not door then return nil end
        for _, name in ipairs({"Door", "DoorPanel", "Panel", "Wood", "Mesh"}) do
            local p = door:FindFirstChild(name, true)
            if p and p:IsA("BasePart") then return p end
        end
        local best, bestVol = nil, 0
        for _, d in ipairs(door:GetDescendants()) do
            if d:IsA("BasePart") then
                local vol = d.Size.X * d.Size.Y * d.Size.Z
                if vol > bestVol then
                    bestVol = vol
                    best = d
                end
            end
        end
        if door:IsA("BasePart") then return door end
        return best or door:FindFirstChildWhichIsA("BasePart", true)
    end

    local function angDist(a, b)
        local d = (a - b) % 360
        if d > 180 then d = 360 - d end
        return math.abs(d)
    end

    local function isDoorOpen()
        local swing = getSwingPart()
        if not swing then return false end
        local ox, oy, oz = swing.Orientation.X, swing.Orientation.Y, swing.Orientation.Z

        local closedScore = angDist(ox, 180) + angDist(oy, 0) + angDist(oz, 180)
        local openScore = angDist(ox, 0) + angDist(oy, -75) + angDist(oz, 0)
        local openScore2 = angDist(ox, 0) + angDist(oy, 75) + angDist(oz, 0)
        openScore = math.min(openScore, openScore2)

        if openScore + 25 < closedScore then
            return true
        end
        if closedScore < 40 then
            return false
        end
        if angDist(oy, -75) < 25 or angDist(oy, 75) < 25 then
            return true
        end
        if angDist(ox, 180) < 30 and angDist(oz, 180) < 30 then
            return false
        end
        return false
    end

    local function standOn(att)
        if not att then return end
        local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local pos = att:IsA("Attachment") and att.WorldPosition or (att:IsA("BasePart") and att.Position)
        if not pos then return end
        if currentPlatform and currentPlatform.Parent then
            local tw = TweenService:Create(currentPlatform, TweenInfo.new(0.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                CFrame = CFrame.new(pos.X, pos.Y - 3.2, pos.Z)
            })
            tw:Play()
            tw.Completed:Wait()
        end
        hrp.CFrame = CFrame.new(pos + Vector3.new(0, 1.5, 0))
        task.wait(0.35)
    end

    if isLocked() then
        SetStatus("Puerta ya locked")
        return true
    end

    if isDoorOpen() then
        SetStatus("Cerrando puerta...")
        standOn(doorAttach)
        FireDoorPrompt(doorPrompt)
        task.wait(1.2)
        if isDoorOpen() then
            SetStatus("No cerro — skip reopen")
        else
            SetStatus("Puerta cerrada")
        end
    else
        SetStatus("Puerta ya cerrada")
    end

    if isLocked() then
        SetStatus("Puerta asegurada")
        return true
    end

    if needsLock() and not isDoorOpen() then
        SetStatus("Lock...")
        standOn(lockAttach)
        if isLocked() then
            SetStatus("Puerta asegurada")
            return true
        end
        if needsLock() then
            FireDoorPrompt(lockPrompt)
            task.wait(1)
        end
        SetStatus(isLocked() and "Puerta asegurada" or "Lock intentado")
    elseif isDoorOpen() then
        SetStatus("Puerta abierta — no lock")
    else
        SetStatus("Puerta ok")
    end
    return true
end

local function FirePromptSimple(obj, forceHold)
    if not obj then return false end
    local prompt = obj:FindFirstChildOfClass("ProximityPrompt") or obj:FindFirstChildWhichIsA("ProximityPrompt", true)
    if not prompt then return false end

    pcall(function()
        prompt.Enabled = true
        prompt.MaxActivationDistance = 60
        prompt.RequiresLineOfSight = false

        local hold = 0
        if forceHold and type(forceHold) == "number" and forceHold > 0 then
            hold = forceHold
        elseif prompt.HoldDuration and prompt.HoldDuration > 0 then
            hold = prompt.HoldDuration + 0.2
        end

        if hold > 0 then
            prompt:InputHoldBegin()
            task.wait(hold)
            prompt:InputHoldEnd()
        end
        if fireproximityprompt then
            fireproximityprompt(prompt)
        end
    end)
    return true
end

local function forceClick(gui)
    if not gui then return false end

    local did = false
    if getconnections then
        for _, name in ipairs({"Activated", "MouseButton1Click", "MouseButton1Down", "MouseButton1Up", "InputBegan", "TouchTap"}) do
            pcall(function()
                local sig = gui[name]
                if typeof(sig) == "RBXScriptSignal" then
                    for _, c in ipairs(getconnections(sig)) do
                        pcall(function()
                            if c.Fire then c:Fire() end
                            if c.Function then c.Function() end
                        end)
                        did = true
                    end
                end
            end)
        end
    end

    if not did and firesignal then
        for _, ev in ipairs({"MouseButton1Click", "Activated", "MouseButton1Down", "MouseButton1Up"}) do
            pcall(function()
                local sig = gui[ev]
                if typeof(sig) == "RBXScriptSignal" then
                    firesignal(sig)
                    did = true
                end
            end)
        end
    end

    if not did then
        pcall(function()
            local pos = gui.AbsolutePosition
            local size = gui.AbsoluteSize
            if size.X < 2 or size.Y < 2 then return end
            local inset = GuiService:GetGuiInset()
            local x = pos.X + size.X / 2
            local y = pos.Y + size.Y / 2 + (inset.Y or 0)
            if isMobile then
                VirtualInputManager:SendTouchEvent(x, y, true, game)
                task.wait(0.06)
                VirtualInputManager:SendTouchEvent(x, y, false, game)
            else
                VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 1)
                task.wait(0.06)
                VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 1)
            end
            pcall(function() GuiService.SelectedObject = nil end)
            pcall(function() UserInputService.MouseIconEnabled = true end)
        end)
    end
    return true
end

local function waitFor(fn, timeout)
    timeout = timeout or 10
    local t = tick()
    while tick() - t < timeout do
        local ok, v = pcall(fn)
        if ok and v then return v end
        task.wait(0.1)
    end
    return nil
end

local function SlowTravel(targetPosOrCF, label, riseSeconds)
    riseSeconds = riseSeconds or 4.5
    local targetPos = typeof(targetPosOrCF) == "CFrame" and targetPosOrCF.Position or targetPosOrCF
    if not targetPos then return false end
    if not EnsurePlatform() then return false end
    local isTpDealer = (label == "TP Dealer")

    SetStatus(label or "Viajando...")

    local down = TweenService:Create(currentPlatform, TweenInfo.new(3.2, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut), {
        CFrame = CFrame.new(currentPlatform.Position.X, SafeHeight, currentPlatform.Position.Z)
    })
    down:Play()
    down.Completed:Wait()
    if not isTpDealer and not anyFarmOn() then ForceGoUp() return false end

    local endPos = Vector3.new(targetPos.X, SafeHeight, targetPos.Z)
    while isTpDealer or anyFarmOn() do
        if not currentPlatform or not currentPlatform.Parent then return false end
        if not isTpDealer and not anyFarmOn() then ForceGoUp() return false end
        local current = currentPlatform.Position
        local remaining = (endPos - Vector3.new(current.X, SafeHeight, current.Z)).Magnitude
        if remaining <= 2 then break end

        local direction = (endPos - Vector3.new(current.X, SafeHeight, current.Z)).Unit
        local step = math.min(SPEED * 0.1, remaining)
        currentPlatform.CFrame = CFrame.new(Vector3.new(current.X, SafeHeight, current.Z) + direction * step)
        task.wait(0.1)
    end

    if not isTpDealer and not anyFarmOn() then ForceGoUp() return false end
    currentPlatform.CFrame = CFrame.new(targetPos.X, SafeHeight, targetPos.Z)
    task.wait(0.2)

    SetStatus("Subiendo...")
    local up = TweenService:Create(currentPlatform, TweenInfo.new(riseSeconds, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        CFrame = CFrame.new(targetPos.X, targetPos.Y - 4.3, targetPos.Z)
    })
    up:Play()
    while up.PlaybackState == Enum.PlaybackState.Playing do
        if not isTpDealer and not anyFarmOn() then up:Cancel() ForceGoUp() return false end
        task.wait(0.1)
    end
    if not isTpDealer and not anyFarmOn() then ForceGoUp() return false end
    task.wait(0.35)
    return true
end

GetToolCount = function(name)
    local count = 0
    local lowerName = string.lower(name)
    local function check(container)
        if not container then return end
        for _, t in ipairs(container:GetChildren()) do
            if t:IsA("Tool") and string.find(string.lower(t.Name), lowerName) then
                count += 1
            end
        end
    end
    check(LocalPlayer:FindFirstChild("Backpack"))
    check(LocalPlayer.Character)
    return count
end

local function GetTool(name)
    local lowerName = string.lower(name)
    local function search(container)
        if not container then return nil end
        for _, t in ipairs(container:GetChildren()) do
            if t:IsA("Tool") and string.find(string.lower(t.Name), lowerName) then
                return t
            end
        end
        return nil
    end
    return search(LocalPlayer:FindFirstChild("Backpack")) or search(LocalPlayer.Character)
end

local function SellAllBags()
    SetStatus("Vendiendo bolsas...")
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local order = {
        "Small Marshmallow Bag",
        "Medium Marshmallow Bag",
        "Large Marshmallow Bag",
    }

    local maxLoops = 40
    local failStreak = 0

    for _ = 1, maxLoops do
        if not anyFarmOn() then break end
        local okLoop = pcall(function()
            local bagName = nil
            for _, name in ipairs(order) do
                if GetToolCount(name) > 0 then
                    bagName = name
                    break
                end
            end
            if not bagName then
                failStreak = 99
                return
            end

            local before = GetToolCount(bagName)
            local tool = GetTool(bagName)
            if not tool then
                failStreak += 1
                return
            end

            SetStatus("Vendiendo " .. bagName .. "...")
            pcall(function() hum:EquipTool(tool) end)
            task.wait(0.2)
            HoldSellPrompt()
            task.wait(0.45)

            local after = GetToolCount(bagName)
            local sold = before - after
            if sold > 0 then
                TotalMoneyMade += sold * (BAG_PRICES[bagName] or 0)
                pcall(function()
                    local now = GetPlayerMoney()
                    if now > MoneyBefore then
                        TotalMoneyMade = math.max(TotalMoneyMade, now - MoneyBefore)
                    end
                end)
                failStreak = 0
            else
                failStreak += 1
            end
            CheckLimitCash()
        end)
        if not okLoop then failStreak += 1 end
        if failStreak >= 4 then break end
        task.wait(0.15)
    end

    SetStatus("Venta terminada")
    CheckLimitCash()
end

local function ClickGuiObject(obj)
    if not obj then return false end
    local ok = false
    pcall(function()
        for _, ev in ipairs({"Activated", "MouseButton1Click", "MouseButton1Down"}) do
            local sig = obj[ev]
            if typeof(sig) == "RBXScriptSignal" then
                for _, conn in ipairs(getconnections(sig)) do
                    pcall(function() conn:Fire() end)
                    ok = true
                end
            end
        end
    end)
    pcall(function()
        local pos = obj.AbsolutePosition
        local size = obj.AbsoluteSize
        if size.X < 2 or size.Y < 2 then return end
        local inset = GuiService:GetGuiInset()
        local x = pos.X + size.X / 2
        local y = pos.Y + size.Y / 2 + (inset.Y or 0)
        if isMobile then
            VirtualInputManager:SendTouchEvent(x, y, true, game)
            task.wait(0.06)
            VirtualInputManager:SendTouchEvent(x, y, false, game)
        else
            VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 1)
            task.wait(0.06)
            VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 1)
        end
        pcall(function() GuiService.SelectedObject = nil end)
        ok = true
    end)
    return ok
end

local function WaitPath(root, path, timeout)
    timeout = timeout or 6
    local t0 = tick()
    while tick() - t0 < timeout do
        local cur = root
        local ok = true
        for _, name in ipairs(path) do
            if type(name) == "number" then
                local kids = cur and cur:GetChildren()
                cur = kids and kids[name]
            else
                cur = cur and cur:FindFirstChild(name)
            end
            if not cur then ok = false break end
        end
        if ok and cur then return cur end
        task.wait(0.1)
    end
    return nil
end

local function GoToOwnedCookingPot()
    local apt, aptName, aptFolder = FindOwnedApartment()

        if not BuyApartmentWithPot() then return false end
        justBought = true
        apt, aptName, aptFolder = FindOwnedApartment()
        if not apt then return false end
    end

    local pots = GetCookingPots(apt)

    local farmPot = pots[1]
    if not farmPot then
        SetStatus("No se encontro Cooking Pot")

    end

ApplyStretch = function(value)
    stretchValue = math.clamp(math.floor(value), 30, 100)
    if stretchConnection then stretchConnection:Disconnect() stretchConnection = nil end
    if not stretchOn or stretchValue >= 100 then return end
    local mul = stretchValue / 100
    stretchConnection = RunService.RenderStepped:Connect(function()
        local cam = workspace.CurrentCamera
        if not cam then return end
        local cf = cam.CFrame
        cam.CFrame = CFrame.fromMatrix(cf.Position, cf.RightVector, cf.UpVector * mul, -cf.LookVector)
    end)
end

local function CanDelete(obj)
    if not obj or not obj.Parent then return false end
    if IsProtectedObject(obj) then return false end
    if obj.Size and obj.Size.X > 250 and obj.Size.Z > 250 then return false end
    if obj.Position and obj.Position.Y < -30 then return false end
    return obj:IsA("BasePart") or obj:IsA("MeshPart") or obj:IsA("UnionOperation")
end

    ApproxActive = false
    SetStatus("Iniciando...")
    task.spawn(function()

        while AutoMS do
            local okCycle = pcall(function()
                if firstCycle then
                    if not FindOwnedApartment() then
                        local free = FindFreeCompactWalkup()
                        if not free then
                            SetStatus("Depas Llenos")
                            AutoMS = false
                            return
                        end
                        SetStatus("Sin depa — comprando...")
                        if not BuyApartmentWithPot() then
                            task.wait(2)
                            return
                        end
                        if AutoMS then
                            SetStatus("Casa lista — yendo a Mask...")
                            local okMask, errMask = pcall(BuyMaskAtBot)
                            if not okMask then
                                SetStatus("Mask error: " .. tostring(errMask))
                            end
                        end
                    end
                    if not AutoMS then return end

                    if not SlowTravel(ShopPosition, "Yendo a tienda...", 4.5) then
                        if not AutoMS then return end
                        task.wait(1)
                        return
                    end
                    if not AutoMS then return end
                    DestroyPlatform()
                    ClearBlockersNear(ShopPosition.Position, SHOP_CLEAR_RADIUS)
                    task.wait(0.3)
                    if not AutoMS then return end

                    if not AutoBuyIngredients(farmAmount) then
                        SetStatus("Fallo compra — reintentando...")
                        task.wait(1.5)
                        return
                    end
                    if not AutoMS then return end
                    firstCycle = false
                end

                local ok, farmPot = GoToOwnedCookingPot()
                if not ok then
                    SetStatus("Fallo depa — reintentando...")
                    task.wait(1.5)
                    return
                end
                if not AutoMS then return end

                local potPart = farmPot and GetPotPart(farmPot)

                while AutoMS do
                    CheckLimitCash()
                    if HasNoIngredients() then break end
                    if potPart and potPart.Parent then
                        ClearAroundPot(nil, farmPot, 14)
                        PlaceUnderPot(farmPot)
                    elseif farmPot and farmPot.Parent then
                        potPart = GetPotPart(farmPot)
                    end

                    SetStatus("Poniendo Water...")
                    local usedWater = EquipAndUse2Times("Water", farmPot)
                    if usedWater and not ApproxActive then
                        ApproxActive = true
                        ApproxStartTime = tick()
                    end
                    if not AutoMS then break end
                    GoDownAndUp(18)
                    if not AutoMS then break end
                    PlaceUnderPot(farmPot)
                    SetStatus("Poniendo Sugar...")
                    EquipAndUse2Times("Sugar Block Bag", farmPot)
                    if not AutoMS then break end
                    task.wait(1.3)
                    if not AutoMS then break end
                    SetStatus("Poniendo Gelatin...")
                    EquipAndUse2Times("Gelatin", farmPot)
                    if not AutoMS then break end
                    GoDownAndUp(43)
                    if not AutoMS then break end
                    PlaceUnderPot(farmPot)
                    SetStatus("Usando Empty Bag...")
                    local empty = GetTool("Empty Bag") or GetTool("Empty") or GetTool("Bag")
                    if empty then EquipAndUse2Times(empty.Name, farmPot) end
                    BagsMade = (BagsMade or 0) + 1
                    SetStatus("Ciclo olla... bolsas " .. tostring(BagsMade))
                    task.wait(1.0)
                end
                if not AutoMS then return end

                local boughtAgain = FinishFarmAndSell()
                if not AutoMS then return end
                if not boughtAgain then
                    SetStatus("No se pudo comprar, reintentando...")
                    task.wait(1.5)
                    return
                end
                SetStatus("Yendo al depa despues de comprar...")
                task.wait(0.3)
            end)

            if not okCycle then
                SetStatus("Error ciclo — reintentando...")
                pcall(DestroyPlatform)
                task.wait(1.5)
            end
            if not AutoMS then break end
            if CurrentStatus == "Depas Llenos" then break end
        end

        StopLookAt()
        if isProtecting and currentPlatform then ForceGoUp() end
        if CurrentStatus ~= "Depas Llenos" then SetStatus("Detenido") end
    end)
end

local function BoxInstantInteract(root, radius)
    radius = radius or 35
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end

    local targets = {}
    if root then
        pcall(function()
            for _, inst in ipairs(root:GetDescendants()) do
                table.insert(targets, inst)
            end
        end)
    end
    pcall(function()
        for _, inst in ipairs(workspace:GetDescendants()) do
            if inst:IsA("ProximityPrompt") then
                local part = inst.Parent
                if part and part:IsA("BasePart") and (part.Position - hrp.Position).Magnitude <= radius then
                    table.insert(targets, inst)
                end
            end
        end
    end)

    local did = false
    for _, inst in ipairs(targets) do
        if inst:IsA("ProximityPrompt") then
            pcall(function()
                inst.Enabled = true
                inst.MaxActivationDistance = 120
                inst.RequiresLineOfSight = false
                inst.HoldDuration = 0
                if fireproximityprompt then
                    fireproximityprompt(inst)
                    did = true
                end
                pcall(function()
                    inst:InputHoldBegin()
                    task.wait(0.05)
                    inst:InputHoldEnd()
                end)
            end)
        elseif inst:IsA("ClickDetector") then
            pcall(function()
                if fireclickdetector then
                    fireclickdetector(inst)
                    did = true
                end
            end)
        end
    end

    for _ = 1, 3 do
        pcall(PressE)
        task.wait(0.12)
    end
    return did
end

local function BoxFindTool()
    local names = {"box", "crate", "caja", "package", "parcel"}
    local function scan(container)
        if not container then return nil end
        for _, t in ipairs(container:GetChildren()) do
            if t:IsA("Tool") then
                local n = string.lower(t.Name)
                for _, k in ipairs(names) do
                    if n == k or string.find(n, k, 1, true) then
                        return t
                    end
                end
            end
        end
        return nil
    end
    local char = LocalPlayer.Character
    local bp = LocalPlayer:FindFirstChild("Backpack")
    return scan(char) or scan(bp)
end

end

local function HumanoidWalkCircles(rounds)
    rounds = rounds or 4
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    local anchor = BOX_LOC3.Position
    local baseY = anchor.Y + 3

    pcall(DestroyPlatform)
    isProtecting = false
    pcall(function()
        if stayConnection then stayConnection:Disconnect() stayConnection = nil end
    end)
    pcall(function()
        hum.PlatformStand = false
        hum.Sit = false
        hum.AutoRotate = true
        if hum.WalkSpeed < 14 then hum.WalkSpeed = 16 end
        hrp.Anchored = false
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.CFrame = CFrame.new(anchor.X, baseY, anchor.Z)
    end)
    task.wait(0.2)

    local keys = { Enum.KeyCode.W, Enum.KeyCode.D, Enum.KeyCode.S, Enum.KeyCode.A }
    local dirs = {
        Vector3.new(0, 0, -1),
        Vector3.new(1, 0, 0),
        Vector3.new(0, 0, 1),
        Vector3.new(-1, 0, 0),
    }

    SetStatus("Box: vueltas estables x" .. tostring(rounds))
    for r = 1, rounds do
        if not AutoBoxFarm then return end
        for i = 1, 4 do
            if not AutoBoxFarm then return end
            local key = keys[i]
            local dir = dirs[i]
            pcall(function()
                VirtualInputManager:SendKeyEvent(true, key, false, game)
            end)
            local t0 = tick()
            while tick() - t0 < 0.18 and AutoBoxFarm do
                pcall(function()
                    hum:Move(dir, false)
                    local p = hrp.Position + dir * 0.6
                    local dx, dz = p.X - anchor.X, p.Z - anchor.Z
                    if (dx * dx + dz * dz) > (4 * 4) then
                        p = Vector3.new(anchor.X, baseY, anchor.Z)
                    end
                    hrp.CFrame = CFrame.new(p.X, baseY, p.Z)
                end)
                RunService.Heartbeat:Wait()
            end
            pcall(function()
                VirtualInputManager:SendKeyEvent(false, key, false, game)
                hum:Move(Vector3.zero, false)
            end)
            pcall(function()
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.CFrame = CFrame.new(anchor.X, baseY, anchor.Z)
            end)
            task.wait(0.05)
        end
        task.wait(0.06)
    end

    pcall(function()
        hum:Move(Vector3.zero, false)
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.CFrame = CFrame.new(anchor.X, baseY, anchor.Z)
    end)
    task.wait(0.25)
end

local function GetNameChar()
    local folder = workspace:FindFirstChild("Characters")
    if folder then
        local byName = folder:FindFirstChild(LocalPlayer.Name)
        if byName then return byName end
        for _, ch in ipairs(folder:GetChildren()) do
            if Players:GetPlayerFromCharacter(ch) == LocalPlayer then
                return ch
            end
        end
    end
    return LocalPlayer.Character
end

function ApplyCustomName()
    local char = GetNameChar()
    if not char then return false end
    local head = char:FindFirstChild("Head")
    if not head then return false end
    local ok = false
    if customNameOn then
        local nameTag = head:FindFirstChild("NameTag")
        if nameTag then
            local label = nameTag:FindFirstChild("NameLabel", true)
            if label and (label:IsA("TextLabel") or label:IsA("TextButton") or label:IsA("TextBox")) then
                label.Text = tostring(customNombre)
                ok = true
            end
        end
    end
    if customRankOn then
        local rankTag = head:FindFirstChild("RankTag")
        if rankTag then
            local label = rankTag:FindFirstChild("NameLabel", true)
            if label and (label:IsA("TextLabel") or label:IsA("TextButton") or label:IsA("TextBox")) then
                label.Text = tostring(customMini)
                ok = true
            end
        end
    end
    return ok
end

local GREEN_MONEY = Color3.fromRGB(80, 255, 110)

local function collectLabelInstances(lbl)
    local roots = {}
    local function add(x)
        if typeof(x) == "Instance" then
            table.insert(roots, x)
        elseif type(x) == "table" then
            if x.Instance then table.insert(roots, x.Instance) end
            if x.Items then
                for _, v in pairs(x.Items) do add(v) end
            end
        end
    end
    add(lbl)
    return roots
end

local function paintGreen(lbl)
    pcall(function()
        for _, root in ipairs(collectLabelInstances(lbl)) do
            local list = {root}
            pcall(function()
                for _, d in ipairs(root:GetDescendants()) do
                    table.insert(list, d)
                end
            end)
            for _, inst in ipairs(list) do
                if inst:IsA("TextLabel") or inst:IsA("TextButton") or inst:IsA("TextBox") then
                    pcall(function()
                        inst.RichText = true
                        inst.TextColor3 = GREEN_MONEY
                    end)
                    pcall(function()
                        if Library and Library.ThemeMap then
                            Library.ThemeMap[inst] = nil
                        end
                    end)
                    pcall(function()
                        if Library and Library.ChangeItemTheme then
                        end
                    end)
                end
            end
        end
    end)
end

local function wrapLabel(lbl)
    if type(lbl) ~= "table" then return lbl end
    function lbl:SetText(txt)
        txt = tostring(txt)
        pcall(function()
            if self.Set then self:Set(txt) end
        end)
        pcall(function()
            local roots = {}
            if self.Instance then table.insert(roots, self.Instance) end
            if self.Items then
                for _, v in pairs(self.Items) do
                    if type(v) == "table" and v.Instance then table.insert(roots, v.Instance) end
                end
            end
            for _, root in ipairs(roots) do
                if root.Text ~= nil then root.Text = txt end
                for _, d in ipairs(root:GetDescendants()) do
                    if d:IsA("TextLabel") or d:IsA("TextButton") then
                        d.Text = txt
                    end
                end
            end
        end)
    end
    return lbl
end
